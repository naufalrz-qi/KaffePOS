# M2 App Kasir Flutter — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** App kasir Android (Flutter) yang bisa berjualan penuh tanpa internet: aktivasi device, login PIN, shift & kas, transaksi dengan modifier/diskon/pajak, dan struk Bluetooth. Semua data tersinkron otomatis ke backend M0–M1 lewat outbox.

**Architecture:** Proyek Flutter di `mobile/`.
- Master data disimpan sebagai satu snapshot JSON di SQLite.
- Setiap perubahan (shift, kas, penjualan, void) ditulis ke tabel datanya **dan** ke tabel `outbox` dalam satu transaksi SQLite.
- `SyncEngine` mengirim outbox ke `POST /sync/push`, lalu mengecek `GET /sync/master`.
- `AppState` (ChangeNotifier) adalah satu-satunya pengendali yang dipakai UI.
- Printer lewat MethodChannel Kotlin (Bluetooth SPP) dengan perintah ESC/POS buatan sendiri.
- Backend ditambah: metode bayar aktif per usaha dan gambar QRIS di snapshot.

**Tech Stack:**
- App: Flutter stable terbaru (Dart 3), sqflite, http, flutter_secure_storage, uuid, connectivity_plus, workmanager, share_plus, crypto.
- Test app: flutter_test, sqflite_common_ffi (SQLite in-memory di Windows), `package:http/testing.dart`.
- Backend: Django 5.2 + pytest, sama dengan M0–M1.

**Spec:**
- `docs/PRD.md` v0.3: §8.3 dan §9.
- `docs/DESAIN-TEKNIS.md`: §5–§7 dan §13.
- Kontrak API yang sudah berjalan: `docs/superpowers/plans/2026-11-19-m0-m1-fondasi-sync-backend.md` serta `backend/api/push.py` dan `backend/api/snapshot.py`.

```python
QRIS_MAX_BYTES = 300 * 1024


class OutletForm(forms.ModelForm):
    qris = forms.FileField(label="Gambar QRIS statis", required=False,
                           help_text="PNG/JPG maks 300 KB. Kosongkan jika tidak diganti.")

    class Meta:
        model = Outlet
        fields = ["code", "name", "type", "timezone", "tax_enabled", "tax_rate_bp", "tax_inclusive",
                  "rounding_unit", "void_needs_manager", "receipt_header", "receipt_footer", "is_active"]

    def clean_qris(self):
        upload = self.cleaned_data["qris"]
        if not upload:
            return None
        data = upload.read()
        if len(data) > QRIS_MAX_BYTES:
            raise forms.ValidationError("Ukuran gambar maksimal 300 KB.")
        if not data.startswith((b"\x89PNG", b"\xff\xd8")):
            raise forms.ValidationError("Gambar harus PNG atau JPG.")
        return data

    def save(self, commit=True):
        if self.cleaned_data.get("qris"):
            self.instance.qris_image = self.cleaned_data["qris"]
        return super().save(commit)


class BusinessForm(forms.ModelForm):
    payment_methods = forms.MultipleChoiceField(label="Metode pembayaran aktif", choices=PAYMENT_METHODS,
                                                widget=forms.CheckboxSelectMultiple)

    class Meta:
        model = Tenant
        fields = ["name", "receipt_footer"]

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.initial["payment_methods"] = self.instance.payment_methods.split(",")

    def save(self, commit=True):
        chosen = self.cleaned_data["payment_methods"]
        self.instance.payment_methods = ",".join(code for code, _ in PAYMENT_METHODS if code in chosen)
        return super().save(commit)
```

```dart
/// Rumus total transaksi (desain §7.1). Harus identik dengan backend/sales/totals.py;
/// keduanya diuji dengan shared/totals_cases.json.
class Totals {
  const Totals(this.subtotal, this.taxTotal, this.rounding, this.total);

  final int subtotal, taxTotal, rounding, total;

  Map<String, int> toJson() => {'subtotal': subtotal, 'tax_total': taxTotal, 'rounding': rounding, 'total': total};
}

/// Pembagian bulat half-up untuk a ≥ 0 dan b > 0.
int halfUp(int a, int b) => (2 * a + b) ~/ (2 * b);

/// [items] berbentuk item payload `sale.completed`: unit_price, qty, discount, modifiers[].price_delta.
Totals computeTotals(List<Map<String, dynamic>> items, int orderDiscount,
    {required bool taxEnabled, required int taxRateBp, required bool taxInclusive, required int roundingUnit}) {
  var subtotal = 0;
  for (final item in items) {
    final deltas = (item['modifiers'] as List).fold<int>(0, (sum, m) => sum + (m['price_delta'] as int));
    subtotal += ((item['unit_price'] as int) + deltas) * (item['qty'] as int) - (item['discount'] as int);
  }
  final net = subtotal - orderDiscount;
  final tax = !taxEnabled
      ? 0
      : taxInclusive
          ? halfUp(net * taxRateBp, 10000 + taxRateBp)
          : halfUp(net * taxRateBp, 10000);
  final before = taxEnabled && !taxInclusive ? net + tax : net;
  final total = roundingUnit > 0 ? before ~/ roundingUnit * roundingUnit : before;
  return Totals(subtotal, tax, total - before, total);
}
```

```dart
import 'dart:convert';
import 'dart:isolate';

import 'package:crypto/crypto.dart';

/// PBKDF2-HMAC-SHA256 dengan output 32 byte (satu blok), sama dengan hasher Django.
List<int> pbkdf2Sha256(List<int> password, List<int> salt, int iterations) {
  final hmac = Hmac(sha256, password);
  var u = hmac.convert([...salt, 0, 0, 0, 1]).bytes;
  final out = List<int>.of(u);
  for (var i = 1; i < iterations; i++) {
    u = hmac.convert(u).bytes;
    for (var j = 0; j < out.length; j++) {
      out[j] ^= u[j];
    }
  }
  return out;
}

/// Cocokkan PIN dengan hash format Django `pbkdf2_sha256$iterasi$salt$hash` (desain §3.3).
bool checkPinSync(String pin, String encoded) {
  final parts = encoded.split(r'$');
  if (parts.length != 4 || parts[0] != 'pbkdf2_sha256') return false;
  final iterations = int.tryParse(parts[1]);
  if (iterations == null || iterations < 1) return false;
  return base64.encode(pbkdf2Sha256(utf8.encode(pin), utf8.encode(parts[2]), iterations)) == parts[3];
}

/// Verifikasi di isolate lain supaya UI tidak macet. 20.000 iterasi memakan ratusan ms di HP murah;
/// jumlah iterasi adalah kenop kalibrasi di server (core/pins.py PIN_ITERATIONS).
Future<bool> checkPin(String pin, String encoded) => Isolate.run(() => checkPinSync(pin, encoded));
```

```dart
// Format angka, waktu, dan label untuk layar & struk.

/// 18000 → "18.000"; -41 → "-41".
String ribuan(int value) {
  final digits = value.abs().toString();
  final out = StringBuffer(value < 0 ? '-' : '');
  for (var i = 0; i < digits.length; i++) {
    if (i > 0 && (digits.length - i) % 3 == 0) out.write('.');
    out.write(digits[i]);
  }
  return out.toString();
}

/// 18000 → "Rp18.000"; -6000 → "-Rp6.000".
String rp(int value) => value < 0 ? '-Rp${ribuan(-value)}' : 'Rp${ribuan(value)}';

const _offsetJam = {'Asia/Jakarta': 7, 'Asia/Makassar': 8, 'Asia/Jayapura': 9};

/// Jam dinding outlet. Indonesia tanpa DST, jadi cukup offset tetap (WIB/WITA/WIT).
/// Hasilnya DateTime UTC yang angkanya sudah digeser: hanya untuk diformat, jangan dikirim ke server.
DateTime outletTime(DateTime t, String timezone) => t.toUtc().add(Duration(hours: _offsetJam[timezone] ?? 7));

String _dua(int v) => v.toString().padLeft(2, '0');

/// Bagian tanggal nomor struk: 19 Nov 2026 → "261119".

/// "10:15".
String jam(DateTime local) => '${_dua(local.hour)}:${_dua(local.minute)}';

/// Waktu ISO-8601 UTC untuk API dan SQLite.
String isoUtc(DateTime t) => t.toUtc().toIso8601String();
```

```dart
/// Akses SQLite untuk meta, snapshot master, dan outbox. Operasi transaksi ada di sales_store.dart.
class Store {
  Store(this.db);

  final Database db;

  Future<String?> meta(String key) async {
    final rows = await db.query('meta', where: 'key = ?', whereArgs: [key]);
    return rows.isEmpty ? null : rows.first['value'] as String;
  }

  /// Nilai null menghapus key.
  Future<void> setMeta(String key, String? value) async {
    if (value == null) {
      await db.delete('meta', where: 'key = ?', whereArgs: [key]);
    } else {
      await db.insert('meta', {'key': key, 'value': value}, conflictAlgorithm: ConflictAlgorithm.replace);
    }
  }

  Future<Snapshot?> loadSnapshot() async {
    final rows = await db.query('master_snapshot');
    return rows.isEmpty ? null : Snapshot.parse(rows.first['version'] as int, rows.first['json'] as String);
  }

  Future<int> snapshotVersion() async => Sqflite.firstIntValue(await db.rawQuery('SELECT version FROM master_snapshot')) ?? 0;

  /// Ganti snapshot master secara utuh (desain §6.2).
  Future<void> saveSnapshot(int version, Map<String, dynamic> snapshot) => db.insert(
      'master_snapshot', {'id': 1, 'version': version, 'json': jsonEncode(snapshot)},
      conflictAlgorithm: ConflictAlgorithm.replace);

  /// Masukkan event ke outbox. Selalu dipanggil di dalam transaksi yang sama dengan perubahan datanya.
  Future<void> addEvent(DatabaseExecutor tx, String type, Map<String, Object?> data, DateTime now) => tx.insert(
      'outbox', {'event_id': newId(), 'type': type, 'payload': jsonEncode(data), 'created_at': isoUtc(now)});

  Future<int> pendingCount() async =>
      Sqflite.firstIntValue(await db.rawQuery("SELECT COUNT(*) FROM outbox WHERE status = 'PENDING'")) ?? 0;

  Future<String?> oldestPendingAt() async =>
      (await db.rawQuery("SELECT MIN(created_at) AS t FROM outbox WHERE status = 'PENDING'")).first['t'] as String?;

  Future<List<OutboxRow>> pendingBatch(int limit) async =>
      [for (final r in await db.query('outbox', where: "status = 'PENDING'", orderBy: 'seq', limit: limit)) OutboxRow(r)];

  Future<List<OutboxRow>> rejected() async =>
      [for (final r in await db.query('outbox', where: "status = 'REJECTED'", orderBy: 'seq DESC')) OutboxRow(r)];

  /// Terapkan hasil POST /sync/push (desain §5): ok/duplicate → SENT, rejected → REJECTED, retry → tetap antre.
  Future<void> applyResults(List<dynamic> results) => db.transaction((tx) async {
        for (final r in results.cast<Map<String, dynamic>>()) {
          switch (r['status']) {
            case 'ok' || 'duplicate':
              await tx.update('outbox', {'status': 'SENT'}, where: 'event_id = ?', whereArgs: [r['id']]);
            case 'rejected':
              await tx.update('outbox', {'status': 'REJECTED', 'last_error': r['reason']},
                  where: 'event_id = ?', whereArgs: [r['id']]);
            default:
              await tx.rawUpdate(
                  'UPDATE outbox SET attempts = attempts + 1, last_error = ? WHERE event_id = ?', [r['reason'], r['id']]);
          }
        }
      });

  /// Hapus semua data lokal (reset device, PRD K-28).
  Future<void> wipe() => db.transaction((tx) async {
        for (final table in ['meta', 'master_snapshot', 'shifts', 'cash_movements', 'sales', 'outbox', 'receipt_seq']) {
          await tx.delete(table);
        }
      });
}
```

```dart
    required this.cashSales,
    required this.cashIn,
    required this.cashOut,
    required this.salesCount,
    required this.salesTotal,
  });

class SaleRow {

/// Operasi transaksi. Setiap perubahan + event outbox-nya ditulis dalam SATU transaksi SQLite (desain §6.2).
extension SalesStore on Store {
  Future<Shift?> openShift() async {
    final rows = await db.query('shifts', where: 'closed_at IS NULL', limit: 1);
    return rows.isEmpty ? null : Shift(rows.first);
  }

  Future<Shift> startShift({required String staffId, required int openingCash, required DateTime now}) async {
    final row = {'id': newId(), 'staff_id': staffId, 'opened_at': isoUtc(now), 'opening_cash': openingCash};
    await db.transaction((tx) async {
      await tx.insert('shifts', row);
      await addEvent(tx, 'shift.opened',
          {'shift_id': row['id'], 'staff_id': staffId, 'opened_at': row['opened_at'], 'opening_cash': openingCash}, now);
    });
    return Shift(row);
  }

  Future<void> moveCash({
    required String shiftId,
    required String staffId,
    required String direction,
    required int amount,
    required String note,
    required DateTime now,
  }) =>
      db.transaction((tx) async {
        final row = {'id': newId(), 'shift_id': shiftId, 'at': isoUtc(now), 'direction': direction, 'amount': amount, 'note': note};
        await tx.insert('cash_movements', row);
        await addEvent(tx, 'cash.moved', {...row, 'staff_id': staffId}, now);
      });

  /// Simpan penjualan (desain §6.2, §7.2): nomor struk dari `receipt_seq[tanggal lokal]`, baris `sales`,
  /// dan event `sale.completed` dalam satu transaksi. [sale] = payload tanpa `receipt_no`.
  Future<Map<String, dynamic>> saveSale(Map<String, dynamic> sale,
          {required String outletCode, required int deviceNo, required String timezone, required DateTime now}) =>
      db.transaction((tx) async {
        final date = yymmdd(outletTime(now, timezone));
        final last = Sqflite.firstIntValue(await tx.rawQuery('SELECT last FROM receipt_seq WHERE date = ?', [date])) ?? 0;
        await tx.insert('receipt_seq', {'date': date, 'last': last + 1}, conflictAlgorithm: ConflictAlgorithm.replace);
        final seq = (last + 1).toString().padLeft(4, '0');
        final full = {...sale, 'receipt_no': '$outletCode-$deviceNo-$date-$seq'};
        await tx.insert('sales', {
          'id': full['sale_id'],
          'shift_id': full['shift_id'],
          'receipt_no': full['receipt_no'],
          'created_at': full['created_at'],
          'total': full['total'],
          'status': 'COMPLETED',
          'payload': jsonEncode(full),
        });
        await addEvent(tx, 'sale.completed', full, now);
        return full;
      });

  Future<void> voidSale({
    required String saleId,
    required String staffId,
    String? approvedById,
    required String reason,
    required DateTime now,
  }) =>
      db.transaction((tx) async {
        await tx.update('sales', {'status': 'VOIDED'}, where: 'id = ?', whereArgs: [saleId]);
        await addEvent(tx, 'sale.voided',
            {'sale_id': saleId, 'staff_id': staffId, 'approved_by_id': approvedById, 'voided_at': isoUtc(now), 'reason': reason}, now);
      });

  Future<ShiftRecap> recap(Shift shift) async {
    var cashSales = 0, salesTotal = 0, cashIn = 0, cashOut = 0;
    final sales = await db.query('sales', where: "shift_id = ? AND status = 'COMPLETED'", whereArgs: [shift.id]);
    for (final row in sales) {
      final payload = jsonDecode(row['payload'] as String) as Map<String, dynamic>;
      salesTotal += payload['total'] as int;
      for (final p in (payload['payments'] as List).cast<Map<String, dynamic>>()) {
        if (p['method'] == 'CASH') cashSales += p['amount'] as int;
      }
    }
    for (final m in await db.query('cash_movements', where: 'shift_id = ?', whereArgs: [shift.id])) {
      if (m['direction'] == 'IN') {
        cashIn += m['amount'] as int;
      } else {
        cashOut += m['amount'] as int;
      }
    }
    return ShiftRecap(
        openingCash: shift.openingCash, cashSales: cashSales, cashIn: cashIn, cashOut: cashOut,
        salesCount: sales.length, salesTotal: salesTotal);
  }

  Future<void> closeShift({required Shift shift, required String staffId, required int countedCash, required DateTime now}) =>
      db.transaction((tx) async {
        await tx.update('shifts', {'closed_at': isoUtc(now), 'counted_cash': countedCash}, where: 'id = ?', whereArgs: [shift.id]);
        await addEvent(tx, 'shift.closed',
            {'shift_id': shift.id, 'staff_id': staffId, 'closed_at': isoUtc(now), 'counted_cash': countedCash}, now);
      });

  /// Riwayat di HP (PRD K-15): isi tabel `sales` = shift berjalan + 7 hari terakhir.
  Future<List<SaleRow>> history() async =>
      [for (final r in await db.query('sales', orderBy: 'created_at DESC', limit: 500)) SaleRow(r)];

  /// Hapus data terkirim yang lebih tua dari 7 hari (desain §6.2). Penjualan shift yang masih terbuka
  /// dan event yang belum terkirim tidak pernah dihapus.
  Future<void> cleanup(DateTime now) async {
    final cutoff = isoUtc(now.subtract(const Duration(days: 7)));
    await db.delete('outbox', where: "status = 'SENT' AND created_at < ?", whereArgs: [cutoff]);
    await db.delete('sales',
        where: 'created_at < ? AND shift_id NOT IN (SELECT id FROM shifts WHERE closed_at IS NULL)', whereArgs: [cutoff]);
    await db.delete('cash_movements', where: 'shift_id IN (SELECT id FROM shifts WHERE closed_at < ?)', whereArgs: [cutoff]);
    await db.delete('shifts', where: 'closed_at < ?', whereArgs: [cutoff]);
  }
}
```

```dart
/// Satu baris keranjang: produk + opsi modifier yang dipilih.
class CartLine {
  CartLine(this.product, this.options, {this.qty = 1, this.note = ''});

  final Product product;
  final List<ModOption> options;
  int qty;
  int discount = 0;
  String note;

  int get unitPrice => product.price + options.fold(0, (sum, o) => sum + o.priceDelta);
  int get gross => unitPrice * qty;
  int get total => gross - discount;

  bool _sameAs(Product p, List<ModOption> opts) =>
      note.isEmpty &&
      discount == 0 &&
      product.id == p.id &&
      options.map((o) => o.id).join(',') == opts.map((o) => o.id).join(',');
}

/// Nominal diskon dari persen dalam basis poin (desain §7.1): basis × bp // 10000.
int percentOf(int basis, int bp) => basis * bp ~/ 10000;

/// Nama grup wajib yang belum dipilih (PRD M-04).
List<String> missingRequired(List<ModGroup> groups, Map<String, Set<String>> chosen) =>
    [for (final g in groups) if (g.required && (chosen[g.id]?.isEmpty ?? true)) g.name];

List<int> quickAmounts(int total) =>
    {total, for (final unit in [5000, 10000, 20000, 50000, 100000]) (total + unit - 1) ~/ unit * unit}.toList()..sort();

class Cart extends ChangeNotifier {
  final lines = <CartLine>[];
  int orderDiscount = 0;
  String discountReason = '';
  String customerLabel = '';

  bool get isEmpty => lines.isEmpty;
  int get itemCount => lines.fold(0, (sum, l) => sum + l.qty);
  int get subtotal => lines.fold(0, (sum, l) => sum + l.total);

  void add(Product product, List<ModOption> options, {int qty = 1, String note = ''}) {
    final same = note.isEmpty ? lines.where((l) => l._sameAs(product, options)).toList() : <CartLine>[];
    if (same.isEmpty) {
      lines.add(CartLine(product, options, qty: qty, note: note));
    } else {
      same.first.qty += qty;
    }
    _changed();
  }

  void setQty(CartLine line, int qty) {
    if (qty <= 0) {
      lines.remove(line);
    } else {
      line.qty = qty;
      line.discount = math.min(line.discount, line.gross);
    }
    _changed();
  }

  void setNote(CartLine line, String note) {
    line.note = note;
    _changed();
  }

  void setLineDiscount(CartLine line, int amount, String reason) {
    line.discount = math.min(amount, line.gross);
    discountReason = reason;
    _changed();
  }

  void setOrderDiscount(int amount, String reason) {
    orderDiscount = amount;
    discountReason = reason;
    _changed();
  }

  void setCustomer(String label) {
    customerLabel = label;
    notifyListeners();
  }

  void clear() {
    lines.clear();
    orderDiscount = 0;
    discountReason = '';
    customerLabel = '';
    notifyListeners();
  }

  /// Diskon transaksi tidak boleh melebihi subtotal, supaya total tidak negatif.
  void _changed() {
    orderDiscount = math.min(orderDiscount, subtotal);
    notifyListeners();
  }

  Totals totals(Outlet outlet) => _compute([for (final l in lines) _priced(l)], outlet);

  Totals _compute(List<Map<String, dynamic>> items, Outlet o) => computeTotals(items, orderDiscount,
      taxEnabled: o.taxEnabled, taxRateBp: o.taxRateBp, taxInclusive: o.taxInclusive, roundingUnit: o.roundingUnit);

  Map<String, dynamic> _priced(CartLine l) => {
        'unit_price': l.product.price,
        'qty': l.qty,
        'discount': l.discount,
        'modifiers': [
          for (final o in l.options) {'option_id': o.id, 'group_name': o.groupName, 'name': o.name, 'price_delta': o.priceDelta},
        ],
      };

  /// Payload event `sale.completed` (desain §5) tanpa `receipt_no`; nomor struk diisi saat disimpan.
  /// [payment] berisi method, tendered, change, reference; amount = total.
  Map<String, dynamic> salePayload({

    required String shiftId,
    required String staffId,
    required DateTime now,
    required Map<String, Object?> payment,
  }) {
    final items = <Map<String, dynamic>>[
      for (final l in lines)
        {
          'id': newId(),
          'product_id': l.product.id,
          'name': l.product.name,
          'category_name': snap.categoryName(l.product.categoryId),
          'earns_stamp': l.product.earnsStamp,
          'is_reward': false,
          'note': l.note,
          ..._priced(l),
        },
    ];
    final t = _compute(items, snap.outlet);
    final discounted = orderDiscount > 0 || lines.any((l) => l.discount > 0);
    return {
      'sale_id': newId(),
      'shift_id': shiftId,
      'staff_id': staffId,
      'created_at': isoUtc(now),
      'customer_label': customerLabel,
      'member': null,
      'pricing': snap.outlet.pricing,
      'items': items,
      'order_discount': orderDiscount,
      'discount_reason': discounted && discountReason.isNotEmpty ? discountReason : null,
      ...t.toJson(),
      'payments': [
        {'id': newId(), ...payment, 'amount': t.total},
      ],
    };
  }
}
```

```dart
import 'package:http/http.dart' as http;
```

```dart
/// Error dari server berformat {"error": kode, "message": teks} (desain §5).
class ApiError implements Exception {
  ApiError(this.status, this.code, this.message);

  final int status;
  final String code, message;

  @override
  String toString() => '$code: $message';
}

/// Klien API mobile /api/v1. Halaman error non-JSON (mis. 502 dari proxy) muncul sebagai FormatException
/// dan diperlakukan seperti gangguan jaringan oleh SyncEngine.
class Api {
  Api(this.base, {http.Client? client, this.token}) : _client = client ?? http.Client();

  final String base;
  final http.Client _client;
  String? token;

  Map<String, String> get _headers => {
        'Content-Type': 'application/json',
        'X-App-Version': appVersion,
        if (token != null) 'Authorization': 'Bearer $token',
      };

  Future<Map<String, dynamic>> _send(String path, [Object? body]) async {
    final uri = Uri.parse('$base$path');
    final res = await (body == null
            ? _client.get(uri, headers: _headers)
            : _client.post(uri, headers: _headers, body: jsonEncode(body)))
        .timeout(const Duration(seconds: 30));
    final data = jsonDecode(utf8.decode(res.bodyBytes)) as Map<String, dynamic>;
    if (res.statusCode >= 400) {
      throw ApiError(res.statusCode, '${data['error'] ?? 'http_${res.statusCode}'}', '${data['message'] ?? ''}');
    }
    return data;
  }

  /// POST /device/activate (tanpa token).
  Future<Map<String, dynamic>> activate(String code) => _send('/device/activate', {'code': code, 'app_version': appVersion});

  /// GET /sync/master?version=…
  Future<Map<String, dynamic>> master(int version) => _send('/sync/master?version=$version');

  /// POST /sync/push, maksimal 100 event.
  Future<Map<String, dynamic>> push(List<Map<String, Object?>> events, int pendingAfter) =>
      _send('/sync/push', {'pending_after': pendingAfter, 'events': events});
}
```

```dart
import 'dart:math' as math;

import 'package:flutter/foundation.dart';

import '../core/format.dart';
import '../data/store.dart';
import 'api.dart';

enum SyncOutcome { ok, offline, revoked, busy }

/// Satu siklus sync (desain §6.3): kirim outbox dulu (per 100 event), lalu cek versi master.
/// Aman dijalankan bersamaan dari foreground & background karena server idempotent.
class SyncEngine {
  SyncEngine(this.store, this.api, {DateTime Function()? clock}) : _clock = clock ?? DateTime.now;

  final Store store;
  final Api api;
  final DateTime Function() _clock;
  bool _running = false;

  Future<SyncOutcome> sync() async {
    if (_running) return SyncOutcome.busy;
    _running = true;
    try {
      while (true) {
        final batch = await store.pendingBatch(100);
        if (batch.isEmpty) break;
        final pending = await store.pendingCount();
        final res = await api.push([for (final e in batch) e.toEvent()], math.max(0, pending - batch.length));
        final results = res['results'] as List;
        await store.applyResults(results);
        if (results.every((r) => r['status'] == 'retry')) break; // tidak ada kemajuan; coba lagi siklus berikutnya
      }
      final master = await api.master(await store.snapshotVersion());
      if (master['changed'] == true) {
        await store.saveSnapshot(master['version'] as int, master['snapshot'] as Map<String, dynamic>);
      }
      final now = _clock().toUtc();
      await store.setMeta('last_sync_at', isoUtc(now));
      await store.setMeta('clock_skew_ms', '${DateTime.parse(master['server_time'] as String).difference(now).inMilliseconds}');
      await store.setMeta('revoked', null);
      await store.setMeta('last_error', null);
      return SyncOutcome.ok;
    } on ApiError catch (e) {
      if (e.status == 401) {
        await store.setMeta('revoked', e.code);
        return SyncOutcome.revoked;
      }
      await store.setMeta('last_error', '$e');
      return SyncOutcome.offline;
    } catch (e) {
      // Jaringan putus, timeout, atau server mati: diam, coba lagi di pemicu berikutnya (desain §6.3).
      debugPrint('sync gagal: $e');
      await store.setMeta('last_error', '$e');
      return SyncOutcome.offline;
    } finally {
      _running = false;
    }
  }
}
```

```dart
import 'dart:async';

import 'package:flutter/widgets.dart';
import 'package:flutter_secure_storage/flutter_secure_storage.dart';

import '../config.dart';
import '../core/cart.dart';
import '../core/format.dart';
import '../core/pin.dart';
import '../data/db.dart';
import '../data/sales_store.dart';

/// Token device di penyimpanan aman HP (desain §9). Test memakai pengganti di memori.
class TokenStore {
  const TokenStore();

  static final _storage = FlutterSecureStorage();

  Future<String?> read() => _storage.read(key: 'device_token');

  Future<void> write(String? token) =>
      token == null ? _storage.delete(key: 'device_token') : _storage.write(key: 'device_token', value: token);
}

/// -1 jika a < b, 0 jika sama, 1 jika a > b (versi x.y.z).
int compareVersions(String a, String b) {
  List<int> parts(String v) => [for (final p in v.split('.')) int.tryParse(p) ?? 0];
  final pa = parts(a), pb = parts(b);
  for (var i = 0; i < 3; i++) {
    final x = i < pa.length ? pa[i] : 0, y = i < pb.length ? pb[i] : 0;
    if (x != y) return x < y ? -1 : 1;
  }
  return 0;
}

/// Pusat state app kasir. UI membaca field-nya lewat ListenableBuilder dan memanggil method-nya.
class AppState extends ChangeNotifier {
  AppState({
    required this.store,
    required this.api,
    this.tokens = const TokenStore(),
    DateTime Function()? clock,
    this.autoSync = true,
  })  : _clock = clock ?? DateTime.now,
        engine = SyncEngine(store, api, clock: clock);

  /// Buka database HP, bersihkan data lama, dan muat status. Dipanggil sekali di main().
  static Future<AppState> open() async {
    final store = Store(await openDb(await dbPath()));
    await store.cleanup(DateTime.now());
    final state = AppState(store: store, api: Api(apiBase));
    await state.init();
    return state;
  }

  final Store store;
  final Api api;
  final TokenStore tokens;
  final SyncEngine engine;
  final bool autoSync;
  final DateTime Function() _clock;
  final cart = Cart();

  bool activated = false;
  Snapshot? snapshot;
  Staff? staff;
  Shift? shift;

  bool online = false;
  bool revoked = false;
  int pending = 0;
  int rejectedCount = 0;
  DateTime? lastSyncAt;
  DateTime? oldestPending;
  int clockSkewMs = 0;

  DateTime now() => _clock().toUtc();

  Future<void> init() async {
    api.token = await tokens.read();
    activated = api.token != null;
    await load();
  }

  /// Muat ulang snapshot (jika versinya berubah, mis. oleh sync background), shift aktif, dan status sync.
  Future<void> load() async {
    if (snapshot?.version != await store.snapshotVersion()) snapshot = await store.loadSnapshot();
    shift = await store.openShift();
    pending = await store.pendingCount();
    rejectedCount = (await store.rejected()).length;
    oldestPending = DateTime.tryParse(await store.oldestPendingAt() ?? '');
    lastSyncAt = DateTime.tryParse(await store.meta('last_sync_at') ?? '');
    clockSkewMs = int.tryParse(await store.meta('clock_skew_ms') ?? '') ?? 0;
    revoked = await store.meta('revoked') != null;
    notifyListeners();
  }

  // ---- Peringatan (PRD K-27..K-29) & izin
  bool get clockWarning => clockSkewMs.abs() > const Duration(minutes: 5).inMilliseconds;
  bool get stalePending => oldestPending != null && now().difference(oldestPending!) > const Duration(hours: 24);
  bool get needsUpdate => snapshot != null && compareVersions(appVersion, snapshot!.minAppVersion) < 0;
  bool get discountNeedsApproval => !staff!.canDiscount && !staff!.isManager;
  bool get voidNeedsApproval => snapshot!.outlet.voidNeedsManager && !staff!.isManager;

  // ---- Aktivasi (PRD K-01)
  /// Melempar [ApiError] (mis. `invalid_code`) atau error jaringan; UI menampilkan pesannya.
  Future<void> activate(String code) async {
    final res = await api.activate(code.trim().toUpperCase());
    api.token = res['token'] as String;
    await tokens.write(api.token);
    await store.setMeta('revoked', null);
    activated = true;
    await syncNow();
  }

  // ---- PIN (PRD K-02): salah 5× → kunci 5 menit. Status kunci di database supaya tetap berlaku setelah app ditutup.
  Future<String?> _checkWithLock(Future<bool> Function() check) async {
    final lockedUntil = DateTime.tryParse(await store.meta('pin_locked_until') ?? '');
    if (lockedUntil != null && lockedUntil.isAfter(now())) {
      final minutes = (lockedUntil.difference(now()).inSeconds + 59) ~/ 60;
      return 'Terlalu banyak PIN salah. Coba lagi $minutes menit lagi.';
    }
    if (await check()) {
      await store.setMeta('pin_fails', null);
      return null;
    }
    final fails = int.parse(await store.meta('pin_fails') ?? '0') + 1;
    if (fails >= 5) {
      await store.setMeta('pin_fails', null);
      await store.setMeta('pin_locked_until', isoUtc(now().add(const Duration(minutes: 5))));
      return 'PIN salah 5 kali. Login dikunci 5 menit.';
    }
    await store.setMeta('pin_fails', '$fails');
    return 'PIN salah. Sisa ${5 - fails} kesempatan.';
  }

  /// Login staf. Mengembalikan null jika berhasil, atau pesan kesalahan.
  Future<String?> login(Staff s, String pin) async {
    final error = await _checkWithLock(() => checkPin(pin, s.pinHash));
    if (error == null) {
      staff = s;
      notifyListeners();
    }
    return error;
  }

  /// Persetujuan manager/owner lewat PIN untuk diskon & void (PRD K-09, K-14).
  Future<(Staff?, String?)> approve(String pin) async {
    Staff? approver;
    final error = await _checkWithLock(() async {
      for (final m in snapshot!.staff.where((s) => s.isManager)) {
        if (await checkPin(pin, m.pinHash)) {
          approver = m;
          return true;
        }
      }
      return false;
    });
    return (approver, error);
  }

  void logout() {
    staff = null;
    cart.clear();
    notifyListeners();
  }

  // ---- Shift & kas (PRD K-03..K-05)
  Future<void> openShift(int openingCash) async {
    if (needsUpdate) throw StateError('Versi aplikasi terlalu lama; update dulu sebelum buka shift.');
    await store.startShift(staffId: staff!.id, openingCash: openingCash, now: now());
    await _afterWrite();
  }

  Future<void> moveCash(String direction, int amount, String note) async {
    await store.moveCash(shiftId: shift!.id, staffId: staff!.id, direction: direction, amount: amount, note: note, now: now());
    await _afterWrite();
  }

  Future<ShiftRecap> recap() => store.recap(shift!);

  Future<void> closeShift(int countedCash) async {
    await store.closeShift(shift: shift!, staffId: staff!.id, countedCash: countedCash, now: now());
    await _afterWrite();
  }

  // ---- Transaksi (PRD K-10..K-15)
  /// Simpan penjualan ke SQLite (commit) SEBELUM layar "Berhasil" dan struk (PRD K-11).
  Future<Map<String, dynamic>> checkout(Map<String, Object?> payment) async {
    final snap = snapshot!;
    final sale = cart.salePayload(snap: snap, shiftId: shift!.id, staffId: staff!.id, now: now(), payment: payment);
    final saved = await store.saveSale(sale,
        outletCode: snap.outlet.code, deviceNo: snap.deviceNo, timezone: snap.outlet.timezone, now: now());
    cart.clear();
    await _afterWrite();
    return saved;
  }

  Future<void> voidSale(SaleRow sale, String reason, {Staff? approvedBy}) async {
    await store.voidSale(saleId: sale.id, staffId: staff!.id, approvedById: approvedBy?.id, reason: reason, now: now());
    await _afterWrite();
  }

  Future<List<SaleRow>> history() => store.history();

  // ---- Sync (PRD K-26)
  Future<void> _afterWrite() async {
    await load();
    if (autoSync) unawaited(syncNow()); // offline → gagal diam-diam, data tetap di outbox
  }

  Future<SyncOutcome> syncNow() async {
    if (api.token == null) return SyncOutcome.offline;
    final outcome = await engine.sync();
    if (outcome != SyncOutcome.busy) online = outcome == SyncOutcome.ok;
    await load();
    return outcome;
  }

  // ---- Reset device (PRD K-28)
  /// Hapus aktivasi & data lokal. Ditolak selama shift terbuka atau masih ada data belum terkirim,
  /// kecuali device sudah dicabut (datanya memang tidak bisa dikirim lagi).
  Future<bool> resetDevice() async {
    if (!revoked && (pending > 0 || shift != null)) return false;
    await tokens.write(null);
    api.token = null;
    await store.wipe();
    activated = false;
    snapshot = null;
    staff = null;
    online = false;
    cart.clear();
    await load();
    return true;
  }
}
```

```dart
class _ActivationPageState extends State<ActivationPage> {
  final _code = TextEditingController();
  var _busy = false;
  String? _error;

  Future<void> _submit() async {
    setState(() {
      _busy = true;
      _error = null;
    });
    String? error;
    try {
      await widget.state.activate(_code.text);
    } on ApiError catch (e) {
      error = e.message;
    } catch (_) {
      error = 'Tidak bisa terhubung ke server. Periksa internet lalu coba lagi.';

        _busy = false;

    }

  Widget build(BuildContext context) => Scaffold(
        appBar: AppBar(title: const Text('Aktivasi HP kasir')),
        body: ListView(padding: const EdgeInsets.all(24), children: [

  Future<void> _open() async {
    setState(() => _busy = true);
    try {
      await widget.state.openShift(parseMoney(_cash.text)); // app.dart lalu menampilkan layar jualan
    } finally {
      if (mounted) setState(() => _busy = false);
    }

  Future<void> _reset() async {
    final state = widget.state;
    if (state.pending > 0 && !state.revoked) {
      snack(context, 'Masih ada ${state.pending} data belum terkirim. Sync dulu sebelum reset.');
      return;
    }
    final ok = await confirm(
        context,
        'Reset device?',
        state.revoked
            ? 'Device sudah dicabut. ${state.pending} data yang belum terkirim akan hilang.'
            : 'Semua data di HP ini dihapus dan HP harus diaktivasi ulang dengan kode baru.');
    if (ok) await state.resetDevice();
  }

  @override
  Widget build(BuildContext context) {
    final state = widget.state;
    return Scaffold(
      appBar: AppBar(title: Text('Halo, ${state.staff!.name}'), actions: [
        TextButton(onPressed: state.logout, child: const Text('Ganti staf')),
```

```dart
/// Pilih opsi modifier sesuai aturan grup, qty, dan catatan (PRD K-07). Pop dengan [CartLine].
class ModifierSheet extends StatefulWidget {
  const ModifierSheet({super.key, required this.product, required this.groups});

  final Product product;
  final List<ModGroup> groups;

  @override
  State<ModifierSheet> createState() => _ModifierSheetState();
}

class _ModifierSheetState extends State<ModifierSheet> {
  final _chosen = <String, Set<String>>{}; // id grup → id opsi
  final _note = TextEditingController();
  var _qty = 1;

  void _toggle(ModGroup group, ModOption option) => setState(() {
        final set = _chosen.putIfAbsent(group.id, () => <String>{});
        if (set.contains(option.id)) {
          set.remove(option.id);
        } else {
          if (!group.multi) set.clear();
          set.add(option.id);
        }
      });

  void _done() => Navigator.pop(
        context,
        CartLine(
          widget.product,
          [
            for (final g in widget.groups)
              for (final o in g.options)
                if (_chosen[g.id]?.contains(o.id) ?? false) o,
          ],
          qty: _qty,
          note: _note.text.trim(),
        ),
      );

  @override
  Widget build(BuildContext context) {
    final missing = missingRequired(widget.groups, _chosen);
    return Padding(
      padding: EdgeInsets.only(bottom: MediaQuery.of(context).viewInsets.bottom),
      child: ListView(shrinkWrap: true, padding: const EdgeInsets.all(16), children: [
        Text(widget.product.name, style: Theme.of(context).textTheme.titleLarge),
        for (final g in widget.groups) ...[
          const SizedBox(height: 12),
          Text('${g.name}${g.required ? ' (wajib)' : ''}${g.multi ? ' · boleh lebih dari satu' : ''}'),
          Wrap(spacing: 8, children: [
            for (final o in g.options)
              FilterChip(
                label: Text(o.priceDelta > 0 ? '${o.name} +${ribuan(o.priceDelta)}' : o.name),
                selected: _chosen[g.id]?.contains(o.id) ?? false,
                onSelected: (_) => _toggle(g, o),
              ),
          ]),
        ],
        TextField(controller: _note, maxLength: 120, decoration: const InputDecoration(labelText: 'Catatan (opsional)')),

          IconButton.filledTonal(onPressed: _qty > 1 ? () => setState(() => _qty--) : null, icon: const Icon(Icons.remove)),
          Padding(padding: const EdgeInsets.symmetric(horizontal: 16), child: Text('$_qty', style: const TextStyle(fontSize: 24))),
          IconButton.filledTonal(onPressed: () => setState(() => _qty++), icon: const Icon(Icons.add)),
          const Spacer(),
          FilledButton(
            onPressed: missing.isEmpty ? _done : null,
            child: Text(missing.isEmpty ? 'Tambah' : 'Pilih ${missing.join(', ')}'),
```

```dart
          ListTile(leading: const Icon(Icons.delete_outline), title: const Text('Hapus item'), onTap: () => Navigator.pop(ctx, 'delete')),
        ]),
      ),
    );
    if (!context.mounted) return;
    switch (action) {
      case 'note':
        final note = await askText(context, 'Catatan item', initial: line.note);
        if (note != null) cart.setNote(line, note);
      case 'discount':
        await _discount(context, line);
      case 'delete':
        cart.setQty(line, 0);
    }
  }

  /// Diskon item ([line]) atau transaksi (null). Staf tanpa izin diskon butuh PIN manager (PRD K-09).
  Future<void> _discount(BuildContext context, CartLine? line) async {
    if (state.discountNeedsApproval) {
      final pin = await askPin(context, 'PIN manager untuk diskon');
      if (pin == null || !context.mounted) return;
      final (manager, error) = await state.approve(pin);
      if (!context.mounted) return;
      if (manager == null) {
        snack(context, error ?? 'PIN manager salah');
        return;
      }
    }
    final result = await askDiscount(context, line?.gross ?? cart.subtotal);
    if (result == null) return;
    final (amount, reason) = result;
    if (line == null) {
      cart.setOrderDiscount(amount, reason);
    } else {
      cart.setLineDiscount(line, amount, reason);
    }
  }
```

```dart
import 'dart:math' as math;

import '../core/format.dart';
import '../data/sales_store.dart';
import '../data/snapshot.dart';

// Teks struk & rekap (desain §6.4): 58 mm = 32 karakter per baris, 80 mm = 48. Dipakai layar, printer, dan share.

String _fit(String left, String right, int width) {
  final room = math.max(0, width - right.length - 1);
  final l = left.length > room ? left.substring(0, room) : left;
  return '$l${' ' * (width - l.length - right.length)}$right';
}

List<String> _wrap(String text, int width) =>
    [for (var i = 0; i < text.length; i += width) text.substring(i, math.min(i + width, text.length))];

String _center(String text, int width) =>
    text.length >= width ? text.substring(0, width) : '${' ' * ((width - text.length) ~/ 2)}$text';

List<String> _centered(List<String> texts, int width) => [
      for (final t in texts)
        if (t.trim().isNotEmpty)
          for (final w in _wrap(t.trim(), width)) _center(w, width),
    ];

String _percent(int bp) => bp % 100 == 0 ? '${bp ~/ 100}%' : '${bp ~/ 100},${(bp % 100).toString().padLeft(2, '0')}%';

List<String> _itemLines(Map<String, dynamic> item, int width) {
  final mods = (item['modifiers'] as List).cast<Map<String, dynamic>>();
  final unit = (item['unit_price'] as int) + mods.fold<int>(0, (sum, m) => sum + (m['price_delta'] as int));
  final qty = item['qty'] as int;
  final discount = item['discount'] as int;
  final note = item['note'] as String? ?? '';
  return [
    ..._wrap(item['name'] as String, width),
    if (mods.isNotEmpty) ..._wrap('  + ${mods.map((m) => m['name']).join(', ')}', width),
    if (note.isNotEmpty) ..._wrap('  * $note', width),
    _fit('  $qty x ${ribuan(unit)}', ribuan(unit * qty), width),
    if (discount > 0) _fit('  Diskon', ribuan(-discount), width),
  ];
}

/// Struk dari payload `sale.completed`. Pajak memakai `pricing` saat transaksi, bukan setting outlet sekarang.
List<String> receiptLines(Map<String, dynamic> sale, Snapshot snap, {int width = 32, bool voided = false}) {
  final outlet = snap.outlet;
  final pricing = sale['pricing'] as Map<String, dynamic>;
  final pay = (sale['payments'] as List).first as Map<String, dynamic>;
  final rule = '-' * width;
  final staffName = snap.staffById(sale['staff_id'] as String)?.name ?? '-';
  final customer = sale['customer_label'] as String? ?? '';
  final created = outletTime(DateTime.parse(sale['created_at'] as String), outlet.timezone);
  return [
    ..._centered([snap.businessName, outlet.name, outlet.receiptHeader], width),
    if (voided) _center('*** VOID ***', width),
    rule,
    sale['receipt_no'] as String,
    _fit(tanggalJam(created), 'Kasir: $staffName', width),
    if (customer.isNotEmpty) 'Pelanggan: $customer',
    rule,
    for (final item in (sale['items'] as List).cast<Map<String, dynamic>>()) ..._itemLines(item, width),
    rule,
    _fit('Subtotal', ribuan(sale['subtotal'] as int), width),
    if ((sale['order_discount'] as int) > 0) _fit('Diskon', ribuan(-(sale['order_discount'] as int)), width),
    if (pricing['tax_enabled'] == true)
      _fit('${pricing['tax_inclusive'] == true ? 'Termasuk pajak' : 'Pajak'} ${_percent(pricing['tax_rate_bp'] as int)}',
          ribuan(sale['tax_total'] as int), width),
    if ((sale['rounding'] as int) != 0) _fit('Pembulatan', ribuan(sale['rounding'] as int), width),
    _fit('TOTAL', ribuan(sale['total'] as int), width),
    _fit(paymentLabels[pay['method']] ?? '${pay['method']}', ribuan((pay['tendered'] ?? pay['amount']) as int), width),
    if (pay['change'] != null) _fit('Kembali', ribuan(pay['change'] as int), width),
    if (pay['reference'] != null) 'Ref: ${pay['reference']}',
    rule,
    ..._centered([outlet.receiptFooter, snap.businessFooter], width),
  ];
}
```

```dart
const _init = [0x1B, 0x40]; // ESC @: reset printer
const _feedAndCut = [0x0A, 0x0A, 0x0A, 0x1D, 0x56, 0x42, 0x00]; // 3 baris kosong + GS V 66 0 (diabaikan printer tanpa pisau)

/// Baris teks → perintah ESC/POS. Karakter di luar ASCII cetak diganti '?' supaya aman di semua code page.
List<int> escposText(List<String> lines) => [
      ..._init,
      for (final line in lines) ...[
        for (final c in line.codeUnits) c >= 0x20 && c < 0x7F ? c : 0x3F,
        0x0A,
      ],
      ..._feedAndCut,
    ];

/// QR native ESC/POS (GS ( k, model 2, koreksi error M). Dipakai cetak tes (spike) dan struk member di M3.
List<int> escposQr(String data, {int moduleSize = 6}) {
  final bytes = utf8.encode(data);
  final len = bytes.length + 3;
  return [
    0x1B, 0x61, 0x01, // rata tengah
    0x1D, 0x28, 0x6B, 0x04, 0x00, 0x31, 0x41, 0x32, 0x00, // model 2
    0x1D, 0x28, 0x6B, 0x03, 0x00, 0x31, 0x43, moduleSize, // ukuran modul
    0x1D, 0x28, 0x6B, 0x03, 0x00, 0x31, 0x45, 0x31, // koreksi error M
    0x1D, 0x28, 0x6B, len & 0xFF, len >> 8, 0x31, 0x50, 0x30, ...bytes, // simpan data
    0x1D, 0x28, 0x6B, 0x03, 0x00, 0x31, 0x51, 0x30, // cetak
```

```dart
/// Jembatan ke MainActivity.kt: Bluetooth Classic SPP ke printer yang sudah di-pair (desain §13).
class PrinterChannel {
  static const _channel = MethodChannel('kaffepos/printer');

  static Future<List<BtPrinter>> paired() async {
    final list = await _channel.invokeListMethod<Map<Object?, Object?>>('paired') ?? const [];
    return [for (final m in list) BtPrinter('${m['name']}', '${m['address']}')];
  }

  static Future<void> send(String address, List<int> bytes) =>
      _channel.invokeMethod<void>('print', {'address': address, 'bytes': Uint8List.fromList(bytes)});
}

/// Printer pilihan, lebar kertas, dan cetak otomatis; disimpan di tabel meta.
class PrinterSettings extends ChangeNotifier {
  PrinterSettings(this.store);

  final Store store;
  String? address;
  String? name;
  int width = 32;
  bool autoPrint = true;

  Future<void> load() async {
    address = await store.meta('printer_address');
    name = await store.meta('printer_name');
    width = int.tryParse(await store.meta('paper_width') ?? '') ?? 32;
    autoPrint = await store.meta('auto_print') != '0';
    notifyListeners();
  }

  Future<void> choose(BtPrinter printer) async {
    await store.setMeta('printer_address', printer.address);
    await store.setMeta('printer_name', printer.name);
    await load();
  }

  Future<void> setWidth(int value) async {
    await store.setMeta('paper_width', '$value');
    await load();
  }

  Future<void> setAutoPrint(bool on) async {
    await store.setMeta('auto_print', on ? '1' : '0');
    await load();
  }

  /// Kirim ke printer. Mengembalikan pesan kesalahan atau null. Gagal cetak tidak pernah membatalkan transaksi (PRD K-13).
  Future<String?> printBytes(List<int> bytes) async {
    final target = address;
    if (target == null) return 'Printer belum dipilih (menu Printer).';
    try {
      await PrinterChannel.send(target, bytes);
      return null;
    } on PlatformException catch (e) {
      return e.code == 'permission'
          ? 'Izinkan akses Bluetooth (Perangkat sekitar), lalu cetak ulang.'
          : 'Printer gagal (${e.message}). Transaksi tetap tersimpan; cetak ulang dari Riwayat.';
    }
  }

  Future<String?> printLines(List<String> lines) => printBytes(escposText(lines));
}
```

```kotlin
package id.kaffepos.kasir

import android.Manifest
import android.annotation.SuppressLint
import android.bluetooth.BluetoothAdapter
import android.bluetooth.BluetoothManager
import android.bluetooth.BluetoothSocket
import android.content.pm.PackageManager
import android.os.Build
import io.flutter.embedding.android.FlutterActivity
import io.flutter.embedding.engine.FlutterEngine
import io.flutter.plugin.common.MethodChannel
import java.io.IOException
import java.util.UUID
import java.util.concurrent.Executors

/** Printer thermal Bluetooth Classic (SPP) untuk channel "kaffepos/printer" (desain §13). */
@SuppressLint("MissingPermission") // izin BLUETOOTH_CONNECT dicek di awal setiap panggilan
class MainActivity : FlutterActivity() {
    private val spp: UUID = UUID.fromString("00001101-0000-1000-8000-00805F9B34FB")
    private val io = Executors.newSingleThreadExecutor()
    private var socket: BluetoothSocket? = null
    private var socketAddress: String? = null

    override fun configureFlutterEngine(flutterEngine: FlutterEngine) {
        super.configureFlutterEngine(flutterEngine)
        MethodChannel(flutterEngine.dartExecutor.binaryMessenger, "kaffepos/printer").setMethodCallHandler { call, result ->
            if (Build.VERSION.SDK_INT >= 31 &&
                checkSelfPermission(Manifest.permission.BLUETOOTH_CONNECT) != PackageManager.PERMISSION_GRANTED
            ) {
                requestPermissions(arrayOf(Manifest.permission.BLUETOOTH_CONNECT), 1)
                result.error("permission", "Izin Bluetooth belum diberikan", null)
                return@setMethodCallHandler
            }
            val adapter = (getSystemService(BLUETOOTH_SERVICE) as BluetoothManager).adapter
            if (adapter == null || !adapter.isEnabled) {
                result.error("io", "Bluetooth mati", null)
                return@setMethodCallHandler
            }
            when (call.method) {
                "paired" -> result.success(adapter.bondedDevices.map {
                    mapOf("name" to (it.name ?: it.address), "address" to it.address)
                })
                "print" -> {
                    val address = call.argument<String>("address")!!
                    val bytes = call.argument<ByteArray>("bytes")!!
                    io.execute {
                        val error = try {
                            write(adapter, address, bytes)
                            null
                        } catch (e: IOException) {
                            e.message ?: "gagal mengirim"
                        }
                        runOnUiThread {
                            if (error == null) result.success(true) else result.error("io", error, null)
                        }
                    }
                }
                else -> result.notImplemented()
            }
        }
    }

    /** Pakai ulang koneksi; jika putus, sambung ulang sekali lalu kirim lagi (PRD K-13: auto-reconnect). */
    private fun write(adapter: BluetoothAdapter, address: String, bytes: ByteArray) {
        for (attempt in 1..2) {
            try {
                val s = socket?.takeIf { socketAddress == address && it.isConnected } ?: connect(adapter, address)
                s.outputStream.write(bytes)
                s.outputStream.flush()
                return
            } catch (e: IOException) {
                closeSocket()
                if (attempt == 2) throw e
            }
        }
    }

    private fun connect(adapter: BluetoothAdapter, address: String): BluetoothSocket {
        closeSocket()
        val device = adapter.getRemoteDevice(address)
        val secure = device.createRfcommSocketToServiceRecord(spp)
        val s = try {
            secure.connect()
            secure
        } catch (e: IOException) {
            try {
                secure.close()
            } catch (ignored: IOException) {
            }
            // Sebagian printer murah hanya menerima koneksi insecure.
            device.createInsecureRfcommSocketToServiceRecord(spp).also { it.connect() }
        }
        socket = s
        socketAddress = address
        return s
    }

    private fun closeSocket() {
        try {
            socket?.close()
        } catch (ignored: IOException) {
        }
        socket = null
        socketAddress = null
    }

    override fun onDestroy() {
        closeSocket()
        io.shutdown()
        super.onDestroy()
    }
}
```

```dart
/// Sync ±15 menit saat app di background (PRD K-26, best-effort sesuai batasan Android).

@pragma('vm:entry-point')
void backgroundSync() {

    return true;
  });
```
