**Tech Stack:** Django 5.2 + mssql-django (SQL Server), Django templates + Pico.css 2 (CDN), openpyxl (baru), pytest-django; Flutter (Dart) untuk perubahan kecil di app.

**Spec:** `docs/superpowers/specs/2026-11-26-m4-laporan-design.md`

```python
"""Pencatatan audit log (PRD W-05). Dipanggil eksplisit di view/handler, di transaksi yang sama."""
from django.db import models

from .models import AuditLog

SECRET_FIELDS = {"pin", "new_password", "password", "qris"}


def rupiah(n: int) -> str:
    return f"{n:,}".replace(",", ".")


def record(action, obj=None, detail="", actor=None, device=None):
    AuditLog.objects.create(
        action=action, actor=actor, device=device,
        object_type=obj._meta.model_name if obj is not None else "",
        object_id=str(obj.pk) if obj is not None else "",
        detail=detail[:500])


def _show(value):
    if value is None or value == "":
        return "-"
    if isinstance(value, bool):
        return "Ya" if value else "Tidak"
    if isinstance(value, int):
        return rupiah(value)
    if isinstance(value, (list, tuple, set, models.QuerySet)):
        return ", ".join(sorted(str(v) for v in value)) or "-"
    return str(value)


def form_changes(form) -> str:
    """'Label: lama → baru' untuk setiap field yang berubah; field rahasia hanya 'Label diubah'."""
    parts = []
    for name in form.changed_data:
        label = form.fields[name].label or name
        if name in SECRET_FIELDS:
            parts.append(f"{label} diubah")
            continue
        old = form.initial.get(name)
        parts.append(f"{label}: {_show(_initial(form, name, old))} → {_show(form.cleaned_data.get(name))}")
    return "; ".join(parts)


def _initial(form, name, value):
    """Nilai awal ModelForm berupa pk; ubah ke objek supaya tampil sebagai nama."""
    field = form.fields[name]
    queryset = getattr(field, "queryset", None)
    if queryset is None or value in (None, ""):
        return value
    if isinstance(value, (list, tuple)):
        return list(queryset.filter(pk__in=[getattr(v, "pk", v) for v in value]))
    return queryset.filter(pk=getattr(value, "pk", value)).first()
```

```python
from datetime import timedelta

from django import forms
from django.http import Http404
from django.utils import timezone

from core.models import Outlet, User

MAX_DAYS = 366


def allowed_outlets(user):
    """Satu-satunya sumber pembatasan outlet: owner semua, manager hanya yang ditugaskan."""
    outlets = Outlet.objects.order_by("name")
    if user.role == User.Role.OWNER:
        return outlets
    return outlets.filter(pk__in=user.outlets.values("pk"))


class ReportFilter(forms.Form):
    dari = forms.DateField(label="Dari", widget=forms.DateInput(attrs={"type": "date"}))
    sampai = forms.DateField(label="Sampai", widget=forms.DateInput(attrs={"type": "date"}))
    outlet = forms.ModelChoiceField(queryset=Outlet.all_tenants.none(), required=False, empty_label="Semua outlet")
    staf = forms.ModelChoiceField(queryset=User.objects.none(), required=False, empty_label="Semua staf",
                                  to_field_name="pk")

    def __init__(self, data, user):
        super().__init__(data)
        self.allowed = allowed_outlets(user)
        self.fields["outlet"].queryset = self.allowed
        self.fields["staf"].queryset = User.in_tenant.exclude(role=User.Role.OWNER, pin_hash="").order_by("display_name")
        outlet_id = (data or {}).get("outlet")
        if outlet_id and not self.allowed.filter(pk=outlet_id).exists():
            raise Http404("Outlet tidak ditemukan")

    def clean(self):
        cleaned = super().clean()
        dari, sampai = cleaned.get("dari"), cleaned.get("sampai")
        if dari and sampai:
            if dari > sampai:
                raise forms.ValidationError("Tanggal 'dari' harus sebelum 'sampai'.")
            if (sampai - dari).days + 1 > MAX_DAYS:
                raise forms.ValidationError(f"Rentang maksimal {MAX_DAYS} hari.")
        return cleaned

    @property
    def outlet_ids(self):
        chosen = self.cleaned_data.get("outlet")
        return [chosen.pk] if chosen else list(self.allowed.values_list("pk", flat=True))


def default_dates(today=None):
    """Default filter = hari ini (WIB). Hari ini per zona outlet dipakai di beranda."""

    return {"dari": today.isoformat(), "sampai": today.isoformat()}
```

```python
"""Semua angka laporan & beranda (spec §4). Satu sumber untuk halaman dan Excel, jadi angkanya selalu sama."""
from collections import defaultdict
from datetime import timedelta
from zoneinfo import ZoneInfo

from django.db.models import Count, F, Q, Sum
from django.utils import timezone

from core.models import Device
from sales.models import Payment, Sale, SaleItem, SaleItemModifier, Shift

STALE = timedelta(hours=24)


def now():
    return timezone.now()  # dipisah supaya bisa diganti di test


def _completed(sales):
    return sales.filter(status=Sale.Status.COMPLETED)


def sales_between(outlet_ids, dari, sampai, staff=None):
    sales = Sale.objects.filter(outlet_id__in=outlet_ids, business_date__range=(dari, sampai))
    return sales.filter(staff=staff) if staff else sales


def sales_today(outlets):
    """Hari ini dihitung per zona waktu outlet (WIB/WITA/WIT)."""
    cond = Q(pk__in=[])
    for outlet in outlets:
        cond |= Q(outlet_id=outlet.pk, business_date=now().astimezone(ZoneInfo(outlet.timezone)).date())
    return Sale.objects.filter(cond)


def summary(sales):
    done = _completed(sales).aggregate(omzet=Sum("total"), transaksi=Count("id"), diskon=Sum("discount_total"))
    item = SaleItem.objects.filter(sale__in=_completed(sales)).aggregate(n=Sum("qty"))["n"] or 0
    void = sales.filter(status=Sale.Status.VOIDED).aggregate(n=Count("id"), v=Sum("total"))
    omzet, n = done["omzet"] or 0, done["transaksi"]
    return {"omzet": omzet, "transaksi": n, "item": item, "rata_rata": round(omzet / n) if n else 0,
            "diskon": done["diskon"] or 0, "void_jumlah": void["n"], "void_nilai": void["v"] or 0}


def _group(sales, field, label):
    rows = (_completed(sales).values(field).annotate(omzet=Sum("total"), transaksi=Count("id")).order_by(field))
    return [{"label": label(r[field]), "omzet": r["omzet"], "transaksi": r["transaksi"]} for r in rows]


def by_day(sales):
    return _group(sales, "business_date", lambda d: d)


def by_outlet(sales):
    return sorted(_group(sales, "outlet__name", str), key=lambda r: -r["omzet"])


def by_staff(sales):
    return sorted(_group(sales, "staff__display_name", str), key=lambda r: -r["omzet"])


def by_method(sales):
    labels = dict(Payment.Method.choices)
    rows = (Payment.objects.filter(sale__in=_completed(sales)).values("method")
            .annotate(omzet=Sum("amount"), transaksi=Count("sale", distinct=True)).order_by("-omzet"))
    return [{"label": labels.get(r["method"], r["method"]), "omzet": r["omzet"], "transaksi": r["transaksi"]}
            for r in rows]


def by_hour(sales):
    rows = _group(sales, "local_hour", int)
    top = max((r["omzet"] for r in rows), default=0)
    for r in rows:
        r["persen"] = round(r["omzet"] * 100 / top) if top else 0
    return rows


def _items(sales, field):
    rows = (SaleItem.objects.filter(sale__in=_completed(sales)).values(field)
            .annotate(qty=Sum("qty"), omzet=Sum("line_total")).order_by("-qty", field))
    return [{"label": r[field] or "-", "qty": r["qty"], "omzet": r["omzet"]} for r in rows]


def by_product(sales):
    return _items(sales, "name")


def by_category(sales):
    return _items(sales, "category_name")


def by_modifier(sales):
    rows = (SaleItemModifier.objects.filter(sale_item__sale__in=_completed(sales))
            .values("group_name", "name")
            .annotate(qty=Sum("sale_item__qty"), omzet=Sum(F("price_delta") * F("sale_item__qty")))
            .order_by("-qty", "group_name", "name"))
    return [{"label": f"{r['group_name']}: {r['name']}", "qty": r["qty"], "omzet": r["omzet"]} for r in rows]
```

```python
def shifts_between(outlet_ids, dari, sampai, staff=None):
    shifts = Shift.objects.filter(outlet_id__in=outlet_ids, business_date__range=(dari, sampai))
    if staff:
```

```python
"""Ekspor laporan ke .xlsx (spec §7). Angka disimpan sebagai angka supaya bisa dijumlah di Excel."""
import io

from openpyxl import Workbook
from openpyxl.styles import Font

from . import queries as q

SHEETS = ["Ringkasan", "Harian", "Outlet", "Metode bayar", "Staf", "Per jam", "Produk", "Kategori",
          "Modifier", "Shift", "Void", "Diskon"]
RUPIAH = "#,##0"


def _sheet(wb, title, header, rows, money_cols=()):
    ws = wb.create_sheet(title)
    ws.append(header)
    for cell in ws[1]:
        cell.font = Font(bold=True)
    for row in rows:
        ws.append(list(row))
    for col in money_cols:
        for (cell,) in ws.iter_rows(min_row=2, min_col=col, max_col=col):
            cell.number_format = RUPIAH
    for col in ws.columns:
        ws.column_dimensions[col[0].column_letter].width = max(12, max(len(str(c.value or "")) for c in col) + 2)


def _naive(dt):
    return dt.replace(tzinfo=None) if dt else None  # Excel tidak menyimpan zona waktu


def build_workbook(outlet_ids, dari, sampai, staff=None, judul=""):
    sales = q.sales_between(outlet_ids, dari, sampai, staff)
    s = q.summary(sales)
    wb = Workbook()
    wb.remove(wb.active)
    _sheet(wb, "Ringkasan", ["Keterangan", "Nilai"], [
        ("Laporan", judul), ("Periode", f"{dari:%d/%m/%Y} – {sampai:%d/%m/%Y}"), ("Omzet", s["omzet"]),
        ("Transaksi", s["transaksi"]), ("Item terjual", s["item"]), ("Rata-rata", s["rata_rata"]),
        ("Diskon", s["diskon"]), ("Void (jumlah)", s["void_jumlah"]), ("Void (nilai)", s["void_nilai"])])
    for title, fn, first in (("Harian", q.by_day, "Tanggal"), ("Outlet", q.by_outlet, "Outlet"),
                             ("Metode bayar", q.by_method, "Metode"), ("Staf", q.by_staff, "Staf"),
                             ("Per jam", q.by_hour, "Jam")):
        _sheet(wb, title, [first, "Transaksi", "Omzet"],
               [(r["label"], r["transaksi"], r["omzet"]) for r in fn(sales)], money_cols=(3,))
    for title, fn, first in (("Produk", q.by_product, "Produk"), ("Kategori", q.by_category, "Kategori"),
                             ("Modifier", q.by_modifier, "Modifier")):
        _sheet(wb, title, [first, "Qty", "Nilai item"],
               [(r["label"], r["qty"], r["omzet"]) for r in fn(sales)], money_cols=(3,))
    _sheet(wb, "Shift", ["Outlet", "Staf", "Buka", "Tutup", "Modal", "Kas seharusnya", "Kas fisik", "Selisih"],
           [(r["outlet"], r["staf"], _naive(r["buka"]), _naive(r["tutup"]), r["modal"], r["seharusnya"],
             r["fisik"], r["selisih"]) for r in q.shifts_between(outlet_ids, dari, sampai, staff)],
           money_cols=(5, 6, 7, 8))
    for title, rows in (("Void", q.voids(sales)), ("Diskon", q.discounts(sales))):
        _sheet(wb, title, ["No struk", "Waktu", "Nilai", "Alasan", "Oleh", "Disetujui"],
               [(r["struk"], _naive(r["waktu"]), r["nilai"], r["alasan"], r["oleh"], r["disetujui"]) for r in rows],
               money_cols=(3,))
    out = io.BytesIO()
    wb.save(out)
    return out.getvalue()
```

```python
from django.contrib.auth.mixins import LoginRequiredMixin, UserPassesTestMixin

from core.models import AuditLog, User

TABS = [("laporan", "Penjualan"), ("laporan_produk", "Produk"), ("laporan_shift", "Shift"),
        ("laporan_void", "Void & diskon")]


class ReportAccess(LoginRequiredMixin, UserPassesTestMixin):
    def test_func(self):
        return self.request.user.role in (User.Role.OWNER, User.Role.MANAGER)


class Beranda(ReportAccess, TemplateView):
    template_name = "reports/beranda.html"

    def get_context_data(self, **kwargs):
        ctx = super().get_context_data(**kwargs)
        outlets = list(allowed_outlets(self.request.user))
        ids = [o.pk for o in outlets]
        sales = q.sales_today(outlets)
        ctx.update(outlets=outlets, s=q.summary(sales), per_outlet=q.by_outlet(sales), per_metode=q.by_method(sales),
                   per_jam=q.by_hour(sales), terlaris=q.by_product(sales)[:5], devices=q.device_status(ids),
                   shift_lama=q.stale_shifts(ids))
        return ctx


class Laporan(ReportAccess, TemplateView):
    template_name = "reports/laporan.html"
    tab = "laporan"

    def get_context_data(self, **kwargs):
        ctx = super().get_context_data(**kwargs)
        data = self.request.GET.copy()
        for k, v in default_dates().items():
            data.setdefault(k, v)
        form = ReportFilter(data, user=self.request.user)
        today = timezone.localdate()
        ctx.update(form=form, tab=self.tab, tabs=TABS, presets=presets(today), query=data.urlencode(),
                   has_outlets=form.allowed.exists())
        if form.is_valid() and ctx["has_outlets"]:
            c = form.cleaned_data
            sales = q.sales_between(form.outlet_ids, c["dari"], c["sampai"], c["staf"])
            ctx["s"] = q.summary(sales)
            if self.tab == "laporan":
                ctx.update(harian=q.by_day(sales), per_outlet=q.by_outlet(sales), per_metode=q.by_method(sales),
                           per_staf=q.by_staff(sales), per_jam=q.by_hour(sales))
            elif self.tab == "laporan_produk":
                ctx.update(produk=q.by_product(sales), kategori=q.by_category(sales), modifier=q.by_modifier(sales))
            elif self.tab == "laporan_shift":
                ctx["shifts"] = q.shifts_between(form.outlet_ids, c["dari"], c["sampai"], c["staf"])
            else:
                ctx.update(voids=q.voids(sales), discounts=q.discounts(sales))
        return ctx


class Excel(ReportAccess, TemplateView):
    def get(self, request):
        data = request.GET.copy()
        for k, v in default_dates().items():
            data.setdefault(k, v)
        form = ReportFilter(data, user=request.user)
        if not form.is_valid():
            return HttpResponse("Filter tidak valid", status=400)
        c = form.cleaned_data
        body = build_workbook(form.outlet_ids, c["dari"], c["sampai"], c["staf"], judul=request.user.tenant.name)
        response = HttpResponse(body, content_type="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet")
        response["Content-Disposition"] = f'attachment; filename="laporan-{c["dari"]}_{c["sampai"]}.xlsx"'
        return response


class Audit(OwnerRequired, ListView):
    template_name = "reports/audit.html"
    paginate_by = 50

    def get_queryset(self):
        logs = AuditLog.objects.select_related("actor", "device").order_by("-at")
        action = self.request.GET.get("aksi")
        return logs.filter(action=action) if action in AuditLog.Action.values else logs

    def get_context_data(self, **kwargs):
        return super().get_context_data(actions=AuditLog.Action.choices, aksi=self.request.GET.get("aksi", ""),
                                        **kwargs)
```
